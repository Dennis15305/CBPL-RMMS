// ==================== DECLARATION ====================

PROGRAM SR_Main
VAR
    xHscFirstCycle : BOOL := TRUE;                     // Первый цикл: синхронно сбрасывает и запускает шесть HSC.
    a_dwHscRaw     : ARRAY[1..6] OF DWORD;             // Сырые выходы CurrentValue аппаратных HSC.
    iHsc           : INT;                              // Индекс обработки шести HSC.
    iRoll          : INT;                              // Индекс обработки пяти раскатов.
    iDwinWord      : INT;                              // Индекс копирования общей части пакета DWIN.

    a_fbHscTracker     : ARRAY[1..6] OF FB_HscTracker;     // Трекеры приращений шести HSC.
    a_fbSpliceFilter   : ARRAY[1..5] OF FB_InputFilter;    // Фильтры кнопок SPLICE.
    a_fbResetFilter    : ARRAY[1..5] OF FB_InputFilter;    // Фильтры кнопок RESET.
    a_fbRollCounter    : ARRAY[1..5] OF FB_RollCounter;    // Счётчики текущего и предыдущего рулона.
    a_fbDwinDataPacker : ARRAY[1..5] OF FB_DwinDataPacker; // Упаковщики значений для пяти DWIN.
    fbLineMeter        : FB_LineMeter;                     // Скорость и общий метраж шестого энкодера.
    fbShiftManager     : FB_ShiftManager;                  // Определение текущей смены по RTC.
    fbStopMachine      : FB_StopMachine;                   // Изолированная логика Стопмашины линии ЛГК.
    fbStopJournalQueue : FB_StopJournalQueue;              // Независимая постоянная очередь смены состояний ЛГК.
    xStopJournalFirstCycle : BOOL := TRUE;
    a_xStopJournalEnqueue : ARRAY[1..2] OF BOOL;
    a_stStopJournalEnqueue : ARRAY[1..2] OF ST_StopJournalEvent;
    stStopBeforeUpdate : ST_StopMachineState;
    stStopEvent : ST_StopJournalEvent;
    stEmptyStopJournalEvent : ST_StopJournalEvent;
    wStopPreviousState : WORD;
    wStopPreviousShift : WORD;
    udiStopPreviousOrderDbId : UDINT;
    udiStopAckEventId : UDINT;
    iStopJournalPersistentIndex : INT;
    fbInterfaceHeartbeat : TON;                           // Секундный heartbeat интерфейсов HMI.
    a_fbHaiwellActivityTimeout : ARRAY[1..5] OF TON;       // Удержание признака движения между опросами Haiwell.
    a_fbHaiwellResetTimeout    : ARRAY[1..5] OF TON;       // Удержание индикации RESET на Haiwell.
    a_fbHaiwellAttentionFlash  : ARRAY[1..5] OF FB_ThreeFlash; // Три жёлтые вспышки после SPLICE или RESET.
    a_xHaiwellActivitySeen     : ARRAY[1..5] OF BOOL;      // На канале недавно были импульсы.
    a_xHaiwellResetSeen        : ARRAY[1..5] OF BOOL;      // На канале недавно был RESET.

    a_xRollEventRequest  : ARRAY[1..5] OF BOOL;         // Одноцикловые запросы записи завершённых рулонов.
    a_stRollEventRequest : ARRAY[1..5] OF ST_RollEvent; // Данные событий от пяти раскатов до помещения в очередь.
    fbRollEventQueue     : FB_RollEventQueue;            // Постоянная FIFO-очередь событий для службы-сборщика.
    fbEventPacker        : FB_EventPacker;               // Упаковка первого события очереди в 16 Modbus-регистров.
    udiAckEventId        : UDINT;                        // EventId, собранный из двух слов подтверждения сервера.
    iPacketWord          : INT;                          // Индекс копирования 12 слов пакета v2 в GVL_SCADA.
    udiServerOrderDatabaseId    : UDINT;                 // Пакет заказа сервера, собранный из Modbus-слов.
    udiServerOrderTechnicalCard : UDINT;
    udiServerOrderNumber        : UDINT;

    fbCollectorTimeout          : TON;                   // Таймер отсутствия heartbeat службы-сборщика.
    wCollectorHeartbeatPrevious : WORD;                  // Предыдущее значение heartbeat службы-сборщика.
    xCollectorHeartbeatChanged  : BOOL;                  // Одноцикловый признак изменения heartbeat.
    xCollectorHeartbeatSeen     : BOOL;                  // Служба хотя бы один раз изменила heartbeat.
    xCollectorConnected         : BOOL;                  // Heartbeat менялся в течение последних пяти секунд.
    uliHaiwellValueX10          : ULINT;                 // Промежуточное значение длины в 0,1 м.
    udiHaiwellValueX10          : UDINT;                 // Ограниченное 32-битное значение для Modbus.
    uiHaiwellSpeedX10           : UINT;                  // Скорость в 0,1 м/мин.
    uliShiftLengthMm            : ULINT;                 // Промежуточная длина текущей смены, мм.
    uliShiftRawTotalMm          : ULINT;                 // Суммарный расход пяти раскатов за смену, мм.
    udiShiftElapsedSeconds      : UDINT;                 // Прошедшее время текущего сменного интервала.
    rShiftAverageSpeedMPerMin   : REAL;                  // Средняя скорость выпуска с начала сменного интервала.
    xRollPulseThisCycle         : BOOL;                  // За цикл получены достоверные импульсы раската.
    xRollResetThisCycle         : BOOL;                  // За цикл выполнен короткий или учтённый RESET.
    xShiftIntervalExpired       : BOOL;                  // Накопители относятся к прошлому 12-часовому интервалу.
    iPersistentEvent            : INT;                   // Индекс очистки постоянной очереди после смены разметки.
    iPersistentRoll             : INT;                   // Индекс очистки постоянного состояния раскатов.
    stEmptyRollEvent            : ST_RollEvent;          // Нулевой шаблон события для безопасной инициализации.
    stEmptyRollState            : ST_RollState;          // Нулевой шаблон состояния раската.
    stEmptyStopMachine          : ST_StopMachinePersistent; // Нулевой шаблон состояния Стопмашины.
    a_wDwinCommon               : ARRAY[5..12] OF WORD;  // Общие для пяти панелей дата, время, смена и статус RTC.
    a_wDwinRtcCommon            : ARRAY[0..3] OF WORD;   // Команда синхронизации встроенных часов DWIN.
END_VAR


// ==================== IMPLEMENTATION ====================

// ============================================================
// 0. Контроль разметки постоянной памяти
// ============================================================

// При первом запуске этой версии старая PERSISTENT-разметка может содержать
// данные прежнего проекта в новых полях. Несовпадение сигнатуры или версии
// очищает только данные приложения, после чего признак сохраняется.
IF (PersistentVars.dwLayoutSignature <> GVL_Constants.c_dwPersistentLayoutSignature)
   OR (PersistentVars.wLayoutVersion <> GVL_Constants.c_wPersistentLayoutVersion) THEN

    FOR iPersistentEvent := 0 TO 127 DO
        PersistentVars.a_stEventQueue[iPersistentEvent] := stEmptyRollEvent;
    END_FOR;

    FOR iPersistentRoll := 1 TO GVL_Constants.c_iRollChannelCount DO
        PersistentVars.a_stRollState[iPersistentRoll] := stEmptyRollState;
        PersistentVars.a_uliShiftRollPulses[iPersistentRoll] := ULINT#0;
    END_FOR;

    PersistentVars.uiEventQueueHead := UINT#0;
    PersistentVars.uiEventQueueTail := UINT#0;
    PersistentVars.uiEventQueueCount := UINT#0;
    PersistentVars.udiNextEventId := UDINT#1;
    PersistentVars.udiQueueOverflowCount := UDINT#0;
    PersistentVars.stStopMachine := stEmptyStopMachine;
    PersistentVars.usiShiftAccumulatorNumber := USINT#0;
    PersistentVars.udiShiftStartTimestampUtc := UDINT#0;
    PersistentVars.uliShiftOutputPulses := ULINT#0;
    PersistentVars.dwLayoutSignature := GVL_Constants.c_dwPersistentLayoutSignature;
    PersistentVars.wLayoutVersion := GVL_Constants.c_wPersistentLayoutVersion;

END_IF;

// Новый журнал имеет собственную persistent-разметку и не сбрасывает
// существующие счётчики рулонов при первом импорте.
IF PersistentVars.stStopJournal.dwSignature <> DWORD#16#534A4C34 THEN
    FOR iStopJournalPersistentIndex := 0 TO 127 DO
        PersistentVars.stStopJournal.a_stQueue[iStopJournalPersistentIndex] := stEmptyStopJournalEvent;
    END_FOR;
    PersistentVars.stStopJournal.uiHead := UINT#0;
    PersistentVars.stStopJournal.uiTail := UINT#0;
    PersistentVars.stStopJournal.uiCount := UINT#0;
    PersistentVars.stStopJournal.udiNextEventId := UDINT#1;
    PersistentVars.stStopJournal.udiOverflowCount := UDINT#0;
    PersistentVars.stStopJournal.uliLinePulses := ULINT#0;
    PersistentVars.stStopJournal.wLastState := WORD#0;
    PersistentVars.stStopJournal.dwSignature := DWORD#16#534A4C34;
END_IF;

// ============================================================
// 1. Вызов шести аппаратных HSC
// ============================================================

HscSimple_0(
    Enable := TRUE,
    Sync := xHscFirstCycle,
    ACK_Modulo := FALSE,
    HSC_Err => GVL_IO.xHscError[1],
    Validity => GVL_IO.xHscValid[1],
    CurrentValue => a_dwHscRaw[1]
);

HscSimple_1(
    Enable := TRUE,
    Sync := xHscFirstCycle,
    ACK_Modulo := FALSE,
    HSC_Err => GVL_IO.xHscError[2],
    Validity => GVL_IO.xHscValid[2],
    CurrentValue => a_dwHscRaw[2]
);

HscSimple_2(
    Enable := TRUE,
    Sync := xHscFirstCycle,
    ACK_Modulo := FALSE,
    HSC_Err => GVL_IO.xHscError[3],
    Validity => GVL_IO.xHscValid[3],
    CurrentValue => a_dwHscRaw[3]
);

HscSimple_3(
    Enable := TRUE,
    Sync := xHscFirstCycle,
    ACK_Modulo := FALSE,
    HSC_Err => GVL_IO.xHscError[4],
    Validity => GVL_IO.xHscValid[4],
    CurrentValue => a_dwHscRaw[4]
);

HscSimple_4(
    Enable := TRUE,
    Sync := xHscFirstCycle,
    ACK_Modulo := FALSE,
    HSC_Err => GVL_IO.xHscError[5],
    Validity => GVL_IO.xHscValid[5],
    CurrentValue => a_dwHscRaw[5]
);

HscSimple_5(
    Enable := TRUE,
    Sync := xHscFirstCycle,
    ACK_Modulo := FALSE,
    HSC_Err => GVL_IO.xHscError[6],
    Validity => GVL_IO.xHscValid[6],
    CurrentValue => a_dwHscRaw[6]
);


// Sync активен только при первом проходе SR_Main.
xHscFirstCycle := FALSE;


// ============================================================
// 2. Обработка абсолютных значений HSC
// ============================================================

GVL_Diagnostics.stSystem.xAnyHscFault := FALSE;
GVL_Diagnostics.stSystem.xAnyConfigFault := FALSE;


FOR iHsc := 1 TO GVL_Constants.c_iHscChannelCount DO

    GVL_IO.udiHscAbs[iHsc] :=
        DWORD_TO_UDINT(a_dwHscRaw[iHsc]);


    a_fbHscTracker[iHsc](
        i_udiHscAbs := GVL_IO.udiHscAbs[iHsc],
        i_xHscValid := GVL_IO.xHscValid[iHsc],
        i_xHscError := GVL_IO.xHscError[iHsc],
        i_udiCounterRange := GVL_Constants.c_udiHscCounterRange,
        i_udiMaxDeltaPerCycle := GVL_Constants.c_udiMaxPulseDeltaPerCycle,
        iq_stState := GVL_Runtime.a_stHscState[iHsc]
    );


    GVL_Diagnostics.stSystem.a_eHscStatus[iHsc] :=
        GVL_Runtime.a_stHscState[iHsc].eStatus;

    GVL_Diagnostics.stSystem.a_xHscFault[iHsc] :=
        (GVL_Runtime.a_stHscState[iHsc].eStatus <> CHANNEL_STATUS_RUNNING)
        AND
        (GVL_Runtime.a_stHscState[iHsc].eStatus <> CHANNEL_STATUS_STARTING);

    GVL_Diagnostics.stSystem.xAnyHscFault :=
        GVL_Diagnostics.stSystem.xAnyHscFault
        OR GVL_Diagnostics.stSystem.a_xHscFault[iHsc];

    GVL_Diagnostics.stSystem.xAnyConfigFault :=
        GVL_Diagnostics.stSystem.xAnyConfigFault
        OR
        (GVL_Runtime.a_stHscState[iHsc].eStatus = CHANNEL_STATUS_CONFIG_ERROR);

END_FOR;


// ============================================================
// 3. Определение номера текущей смены и времени события
// ============================================================

fbShiftManager(
    i_usiShift1StartHour := GVL_Constants.c_usiShift_1_StartHour,
    i_usiShift2StartHour := GVL_Constants.c_usiShift_2_StartHour,
    i_udiUtcOffsetSeconds := GVL_Constants.c_udiLocalUtcOffsetSeconds,
    iq_stState := GVL_Runtime.stShiftState
);


GVL_Diagnostics.stSystem.xRtcFault :=
    NOT GVL_Runtime.stShiftState.xRtcValid;


// Общая часть пакета всех DWIN. Компоненты времени передаются отдельными
// 16-битными числами, поэтому панели не зависят от порядка слов UDINT.
a_wDwinCommon[5] := UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalYear);
a_wDwinCommon[6] := UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalMonth);
a_wDwinCommon[7] := UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalDay);
a_wDwinCommon[8] := UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalHour);
a_wDwinCommon[9] := UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalMinute);
a_wDwinCommon[10] := UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalSecond);
a_wDwinCommon[11] := USINT_TO_WORD(GVL_Runtime.stShiftState.usiCurrentShift);
IF GVL_Runtime.stShiftState.xRtcValid THEN
    a_wDwinCommon[12] := WORD#1;
ELSE
    a_wDwinCommon[12] := WORD#0;
END_IF;


// Системный RTC DWIN устанавливается записью четырёх слов с VP 009C:
// 5AA5, YYMM, DDHH, MMSS. Год передаётся смещением от 2000.
IF GVL_Runtime.stShiftState.xRtcValid
   AND (GVL_Runtime.stShiftState.uiLocalYear >= UINT#2000)
   AND (GVL_Runtime.stShiftState.uiLocalYear <= UINT#2255)
   AND (GVL_Runtime.stShiftState.uiLocalMonth >= UINT#1)
   AND (GVL_Runtime.stShiftState.uiLocalMonth <= UINT#12)
   AND (GVL_Runtime.stShiftState.uiLocalDay >= UINT#1)
   AND (GVL_Runtime.stShiftState.uiLocalDay <= UINT#31)
   AND (GVL_Runtime.stShiftState.uiLocalHour <= UINT#23)
   AND (GVL_Runtime.stShiftState.uiLocalMinute <= UINT#59)
   AND (GVL_Runtime.stShiftState.uiLocalSecond <= UINT#59) THEN

    a_wDwinRtcCommon[0] := WORD#16#5AA5;
    a_wDwinRtcCommon[1] :=
        SHL(
            UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalYear - UINT#2000),
            8
        )
        OR UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalMonth);
    a_wDwinRtcCommon[2] :=
        SHL(UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalDay), 8)
        OR UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalHour);
    a_wDwinRtcCommon[3] :=
        SHL(UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalMinute), 8)
        OR UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalSecond);

ELSE

    // Нулевая сигнатура запрещает установку DWIN по недостоверному времени.
    a_wDwinRtcCommon[0] := WORD#0;
    a_wDwinRtcCommon[1] := WORD#0;
    a_wDwinRtcCommon[2] := WORD#0;
    a_wDwinRtcCommon[3] := WORD#0;

END_IF;


FOR iDwinWord := 0 TO 3 DO
    GVL_HMI.a_wHmi1RtcSync[iDwinWord] := a_wDwinRtcCommon[iDwinWord];
    GVL_HMI.a_wHmi2RtcSync[iDwinWord] := a_wDwinRtcCommon[iDwinWord];
    GVL_HMI.a_wHmi3RtcSync[iDwinWord] := a_wDwinRtcCommon[iDwinWord];
    GVL_HMI.a_wHmi4RtcSync[iDwinWord] := a_wDwinRtcCommon[iDwinWord];
    GVL_HMI.a_wHmi5RtcSync[iDwinWord] := a_wDwinRtcCommon[iDwinWord];
END_FOR;


// Накопители смены не зависят от сбросов отдельных рулонов. Они используют
// те же проверенные приращения HSC и сохраняются при перезапуске ПЛК.
IF GVL_Runtime.stShiftState.xRtcValid THEN

    xShiftIntervalExpired := FALSE;
    IF PersistentVars.udiShiftStartTimestampUtc <> UDINT#0 THEN
        IF GVL_Runtime.stShiftState.udiTimestampUtc
           < PersistentVars.udiShiftStartTimestampUtc THEN
            xShiftIntervalExpired := TRUE;
        ELSE
            xShiftIntervalExpired :=
                (GVL_Runtime.stShiftState.udiTimestampUtc
                 - PersistentVars.udiShiftStartTimestampUtc)
                >= GVL_Constants.c_udiShiftMaxIntervalSeconds;
        END_IF;
    END_IF;

    IF (PersistentVars.usiShiftAccumulatorNumber < USINT#1)
       OR (PersistentVars.usiShiftAccumulatorNumber > USINT#2)
       OR (PersistentVars.usiShiftAccumulatorNumber <> GVL_Runtime.stShiftState.usiCurrentShift)
       OR (PersistentVars.udiShiftStartTimestampUtc = UDINT#0)
       OR xShiftIntervalExpired THEN

        PersistentVars.usiShiftAccumulatorNumber :=
            GVL_Runtime.stShiftState.usiCurrentShift;
        PersistentVars.udiShiftStartTimestampUtc :=
            GVL_Runtime.stShiftState.udiTimestampUtc;
        PersistentVars.uliShiftOutputPulses := ULINT#0;

        FOR iRoll := 1 TO GVL_Constants.c_iRollChannelCount DO
            PersistentVars.a_uliShiftRollPulses[iRoll] := ULINT#0;
        END_FOR;

    END_IF;

    FOR iRoll := 1 TO GVL_Constants.c_iRollChannelCount DO
        IF GVL_Runtime.a_stHscState[iRoll].xDataValid THEN
            IF PersistentVars.a_uliShiftRollPulses[iRoll]
               <= (ULINT#18446744073709551615
                   - UDINT_TO_ULINT(GVL_Runtime.a_stHscState[iRoll].udiPulseDelta)) THEN
                PersistentVars.a_uliShiftRollPulses[iRoll] :=
                    PersistentVars.a_uliShiftRollPulses[iRoll]
                    + UDINT_TO_ULINT(GVL_Runtime.a_stHscState[iRoll].udiPulseDelta);
            ELSE
                PersistentVars.a_uliShiftRollPulses[iRoll] :=
                    ULINT#18446744073709551615;
                GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
            END_IF;
        END_IF;
    END_FOR;

    IF GVL_Runtime.a_stHscState[6].xDataValid THEN
        IF PersistentVars.uliShiftOutputPulses
           <= (ULINT#18446744073709551615
               - UDINT_TO_ULINT(GVL_Runtime.a_stHscState[6].udiPulseDelta)) THEN
            PersistentVars.uliShiftOutputPulses :=
                PersistentVars.uliShiftOutputPulses
                + UDINT_TO_ULINT(GVL_Runtime.a_stHscState[6].udiPulseDelta);
        ELSE
            PersistentVars.uliShiftOutputPulses := ULINT#18446744073709551615;
            GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
        END_IF;
    END_IF;

END_IF;


// ============================================================
// 4. Фильтрация кнопок и учёт пяти раскатов
// ============================================================

FOR iRoll := 1 TO GVL_Constants.c_iRollChannelCount DO

    a_fbSpliceFilter[iRoll](
        i_xRaw := GVL_IO.xSplice[iRoll],
        i_tOnDelay := GVL_Constants.c_tInputFilter
    );

    a_fbResetFilter[iRoll](
        i_xRaw := GVL_IO.xReset[iRoll],
        i_tOnDelay := GVL_Constants.c_tInputFilter
    );


    a_fbRollCounter[iRoll](
        i_udiPulseDelta := GVL_Runtime.a_stHscState[iRoll].udiPulseDelta,
        i_xPulseDataValid := GVL_Runtime.a_stHscState[iRoll].xDataValid,
        i_xSplice := a_fbSpliceFilter[iRoll].q_xFiltered,
        i_xReset := a_fbResetFilter[iRoll].q_xFiltered,
        i_udiPulsesPerRevolution := GVL_Constants.c_udiEncoderPulsesPerRevolution,
        i_udiWheelCircumferenceMm := GVL_Constants.c_udiWheelCircumferenceMm,
        i_uliManualResetMinLengthMm := GVL_Constants.c_uliManualResetMinLengthMm,
        iq_stState := PersistentVars.a_stRollState[iRoll]
    );


    GVL_Diagnostics.stSystem.xAnyConfigFault :=
        GVL_Diagnostics.stSystem.xAnyConfigFault
        OR PersistentVars.a_stRollState[iRoll].xConfigFault;


    // При завершении рулона создаём неизменяемый снимок для постоянной очереди.
    a_xRollEventRequest[iRoll] :=
        PersistentVars.a_stRollState[iRoll].xRollCompleted;


    // Одноцикловые признаки преобразуем в устойчивые состояния для HMI,
    // которая опрашивает Modbus значительно медленнее цикла программы ПЛК.
    xRollPulseThisCycle :=
        GVL_Runtime.a_stHscState[iRoll].xDataValid
        AND (GVL_Runtime.a_stHscState[iRoll].udiPulseDelta > UDINT#0);

    IF xRollPulseThisCycle THEN
        a_xHaiwellActivitySeen[iRoll] := TRUE;
    END_IF;

    a_fbHaiwellActivityTimeout[iRoll](
        IN := NOT xRollPulseThisCycle,
        PT := GVL_Constants.c_tHaiwellActivityHold
    );

    IF a_fbHaiwellActivityTimeout[iRoll].Q THEN
        a_xHaiwellActivitySeen[iRoll] := FALSE;
    END_IF;

    xRollResetThisCycle :=
        (
            PersistentVars.a_stRollState[iRoll].xShortResetDiscarded
            OR
            (
                PersistentVars.a_stRollState[iRoll].xRollCompleted
                AND (PersistentVars.a_stRollState[iRoll].eLastEndReason = ROLL_END_RESET)
            )
        );

    IF xRollResetThisCycle THEN
        a_xHaiwellResetSeen[iRoll] := TRUE;
    END_IF;

    a_fbHaiwellResetTimeout[iRoll](
        IN := NOT xRollResetThisCycle,
        PT := GVL_Constants.c_tHaiwellResetHold
    );

    IF a_fbHaiwellResetTimeout[iRoll].Q THEN
        a_xHaiwellResetSeen[iRoll] := FALSE;
    END_IF;

    a_fbHaiwellAttentionFlash[iRoll](
        i_xTrigger := a_fbSpliceFilter[iRoll].q_xFiltered
                      OR a_fbResetFilter[iRoll].q_xFiltered,
        i_tHalfPeriod := GVL_Constants.c_tHaiwellFlashHalfPeriod
    );

    IF a_xRollEventRequest[iRoll] THEN
        a_stRollEventRequest[iRoll].udiEventId := UDINT#0; // Настоящий EventId назначит очередь.
        a_stRollEventRequest[iRoll].udiTimestampUtc := GVL_Runtime.stShiftState.udiTimestampUtc;
        a_stRollEventRequest[iRoll].usiUnwindNumber := INT_TO_USINT(iRoll);
        a_stRollEventRequest[iRoll].usiShiftNumber := GVL_Runtime.stShiftState.usiCurrentShift;
        a_stRollEventRequest[iRoll].eEndReason := PersistentVars.a_stRollState[iRoll].eLastEndReason;

        // Физически недостижимый предел защищает явное преобразование ULINT -> UDINT.
        IF PersistentVars.a_stRollState[iRoll].uliPreviousRollLengthMm <= ULINT#4294967295 THEN
            a_stRollEventRequest[iRoll].udiLengthMm :=
                ULINT_TO_UDINT(PersistentVars.a_stRollState[iRoll].uliPreviousRollLengthMm);
        ELSE
            a_stRollEventRequest[iRoll].udiLengthMm := UDINT#4294967295;
            GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
        END_IF;
    END_IF;

    fbStopJournalQueue.q_stHead.wChangeoverSetpointTenthsMin;

GVL_StopJournal.wBreakPoint := fbStopJournalQueue.q_stHead.wBreakPoint;
GVL_StopJournal.wBreakSide := fbStopJournalQueue.q_stHead.wBreakSide;

    UDINT_TO_WORD(fbStopJournalQueue.q_stHead.udiShiftLengthDm MOD UDINT#65536);
GVL_StopJournal.a_wShiftAreaX100[0] :=

    UDINT_TO_WORD(fbStopJournalQueue.q_stHead.udiShiftAreaX100 MOD UDINT#65536);

    GVL_StopJournal.wEventReady := WORD#0;

IF fbStopJournalQueue.q_xStorageFault THEN

ELSIF fbStopJournalQueue.q_xIdExhausted THEN
    GVL_StopJournal.wQueueStatus := WORD#4;
ELSIF PersistentVars.stStopJournal.udiOverflowCount > UDINT#0 THEN

    GVL_StopJournal.wQueueStatus := WORD#1;

    GVL_StopJournal.wQueueStatus := WORD#0;

GVL_StopJournal.a_wOverflowCount[1] :=
    UDINT_TO_WORD(PersistentVars.stStopJournal.udiOverflowCount MOD UDINT#65536);

// Связь со сборщиком считается установленной только после первого изменения
// его heartbeat и пока между изменениями прошло не более пяти секунд.
xCollectorHeartbeatChanged :=
    GVL_SCADA.wCollectorHeartbeat <> wCollectorHeartbeatPrevious;

IF xCollectorHeartbeatChanged THEN
    wCollectorHeartbeatPrevious := GVL_SCADA.wCollectorHeartbeat;
    xCollectorHeartbeatSeen := TRUE;
END_IF;

fbCollectorTimeout(
    IN := NOT xCollectorHeartbeatChanged,
    PT := T#5s
);

xCollectorConnected :=
    xCollectorHeartbeatSeen AND NOT fbCollectorTimeout.Q;


GVL_HAIWELL.wHmiSignature := WORD#16#484D; // ASCII "HM".
GVL_HAIWELL.wHmiProtocolVersion := GVL_Constants.c_wHaiwellProtocolVersion;
GVL_HAIWELL.wPlcHeartbeat := GVL_StopMachine.wPlcHeartbeat;
GVL_HAIWELL.wCurrentShift :=
    USINT_TO_WORD(GVL_Runtime.stShiftState.usiCurrentShift);
GVL_HAIWELL.wLocalHour :=
    UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalHour);
GVL_HAIWELL.wLocalMinute :=
    UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalMinute);
GVL_HAIWELL.wLocalYear :=
    UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalYear);
GVL_HAIWELL.wLocalMonth :=
    UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalMonth);
GVL_HAIWELL.wLocalDay :=
    UINT_TO_WORD(GVL_Runtime.stShiftState.uiLocalDay);
GVL_HAIWELL.wEventQueueCount := GVL_SCADA.wQueueCount;
GVL_HAIWELL.wEventQueueStatus := GVL_SCADA.wQueueStatus;
GVL_HAIWELL.wLastAckResult := GVL_SCADA.wLastAckResult;


// Системные биты: активность PLC, RTC, сборщик, очередь и конфигурация.
GVL_HAIWELL.wSystemStatusBits := WORD#16#0001;
IF GVL_Runtime.stShiftState.xRtcValid THEN
    GVL_HAIWELL.wSystemStatusBits :=
        GVL_HAIWELL.wSystemStatusBits OR WORD#16#0002;
END_IF;
IF xCollectorConnected THEN
    GVL_HAIWELL.wSystemStatusBits :=
        GVL_HAIWELL.wSystemStatusBits OR WORD#16#0004;
END_IF;
IF (GVL_SCADA.wQueueStatus = WORD#3)
   OR (GVL_SCADA.wQueueStatus = WORD#4) THEN
    GVL_HAIWELL.wSystemStatusBits :=
        GVL_HAIWELL.wSystemStatusBits OR WORD#16#0008;
END_IF;
IF GVL_SCADA.wQueueStatus = WORD#2 THEN
    GVL_HAIWELL.wSystemStatusBits :=
        GVL_HAIWELL.wSystemStatusBits OR WORD#16#0010;
END_IF;
IF GVL_Diagnostics.stSystem.xAnyConfigFault
   OR GVL_Diagnostics.stSystem.xAnyHscFault THEN
    GVL_HAIWELL.wSystemStatusBits :=
        GVL_HAIWELL.wSystemStatusBits OR WORD#16#0020;
END_IF;


uliShiftRawTotalMm := ULINT#0;

FOR iRoll := 1 TO GVL_Constants.c_iRollChannelCount DO
    // Текущий и предыдущий рулон уже безопасно упакованы для DWIN.
    GVL_HAIWELL.a_wCurrentLengthX10[iRoll, 0] :=
        a_fbDwinDataPacker[iRoll].q_wCurrentHigh;
    GVL_HAIWELL.a_wCurrentLengthX10[iRoll, 1] :=
        a_fbDwinDataPacker[iRoll].q_wCurrentLow;
    GVL_HAIWELL.a_wPreviousLengthX10[iRoll, 0] :=
        a_fbDwinDataPacker[iRoll].q_wPreviousHigh;
    GVL_HAIWELL.a_wPreviousLengthX10[iRoll, 1] :=
        a_fbDwinDataPacker[iRoll].q_wPreviousLow;

    GVL_HAIWELL.a_wUnwindStatus[iRoll] := WORD#0;
    IF NOT PersistentVars.a_stRollState[iRoll].xConfigFault THEN
        GVL_HAIWELL.a_wUnwindStatus[iRoll] :=
            GVL_HAIWELL.a_wUnwindStatus[iRoll] OR WORD#16#0001;
    END_IF;
    IF a_xHaiwellActivitySeen[iRoll] THEN
        GVL_HAIWELL.a_wUnwindStatus[iRoll] :=
            GVL_HAIWELL.a_wUnwindStatus[iRoll] OR WORD#16#0002;
    END_IF;
    IF a_fbSpliceFilter[iRoll].q_xFiltered THEN
        GVL_HAIWELL.a_wUnwindStatus[iRoll] :=
            GVL_HAIWELL.a_wUnwindStatus[iRoll] OR WORD#16#0004;
    END_IF;
    IF a_xHaiwellResetSeen[iRoll] THEN
        GVL_HAIWELL.a_wUnwindStatus[iRoll] :=
            GVL_HAIWELL.a_wUnwindStatus[iRoll] OR WORD#16#0008;
    END_IF;
    IF PersistentVars.a_stRollState[iRoll].xConfigFault THEN
        GVL_HAIWELL.a_wUnwindStatus[iRoll] :=
            GVL_HAIWELL.a_wUnwindStatus[iRoll] OR WORD#16#0010;
    END_IF;

    // Состояние строки обзора: жёлтая полуволна трёхкратной индикации
    // SPLICE/RESET имеет приоритет над признаком движения.
    IF a_fbHaiwellAttentionFlash[iRoll].q_xFlashOn THEN
        GVL_HAIWELL.a_wUnwindDisplayState[iRoll] := WORD#2;
    ELSIF a_xHaiwellActivitySeen[iRoll] THEN
        GVL_HAIWELL.a_wUnwindDisplayState[iRoll] := WORD#1;
    ELSE
        GVL_HAIWELL.a_wUnwindDisplayState[iRoll] := WORD#0;
    END_IF;

    CASE PersistentVars.a_stRollState[iRoll].eLastEndReason OF
        ROLL_END_SPLICE:
            GVL_HAIWELL.a_wLastEndReason[iRoll] := WORD#1;
        ROLL_END_RESET:
            GVL_HAIWELL.a_wLastEndReason[iRoll] := WORD#2;
        ELSE
            GVL_HAIWELL.a_wLastEndReason[iRoll] := WORD#0;
    END_CASE;

    // Отброшенная длина: миллиметры -> 0,1 м, с насыщением UDINT.
    uliHaiwellValueX10 :=
        PersistentVars.a_stRollState[iRoll].uliDiscardedTotalLengthMm / ULINT#100;
    IF uliHaiwellValueX10 <= ULINT#4294967295 THEN
        udiHaiwellValueX10 := ULINT_TO_UDINT(uliHaiwellValueX10);
    ELSE
        udiHaiwellValueX10 := UDINT#4294967295;
        GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
        GVL_HAIWELL.wSystemStatusBits :=
            GVL_HAIWELL.wSystemStatusBits OR WORD#16#0020;
    END_IF;
    GVL_HAIWELL.a_wDiscardedLengthX10[iRoll, 0] :=
        UDINT_TO_WORD(udiHaiwellValueX10 / UDINT#65536);
    GVL_HAIWELL.a_wDiscardedLengthX10[iRoll, 1] :=
        UDINT_TO_WORD(udiHaiwellValueX10 MOD UDINT#65536);


    // Расход сырья этого раската за смену: импульсы -> мм -> 0,1 м.
    uliShiftLengthMm :=
        (PersistentVars.a_uliShiftRollPulses[iRoll]
         * UDINT_TO_ULINT(GVL_Constants.c_udiWheelCircumferenceMm))
        / UDINT_TO_ULINT(GVL_Constants.c_udiEncoderPulsesPerRevolution);

    IF uliShiftRawTotalMm
       <= (ULINT#18446744073709551615 - uliShiftLengthMm) THEN
        uliShiftRawTotalMm := uliShiftRawTotalMm + uliShiftLengthMm;
    ELSE
        uliShiftRawTotalMm := ULINT#18446744073709551615;
        GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
    END_IF;

    uliHaiwellValueX10 := uliShiftLengthMm / ULINT#100;
    IF uliHaiwellValueX10 <= ULINT#4294967295 THEN
        udiHaiwellValueX10 := ULINT_TO_UDINT(uliHaiwellValueX10);
    ELSE
        udiHaiwellValueX10 := UDINT#4294967295;
        GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
    END_IF;
    GVL_HAIWELL.a_wShiftRollLengthX10[iRoll, 0] :=
        UDINT_TO_WORD(udiHaiwellValueX10 / UDINT#65536);
    GVL_HAIWELL.a_wShiftRollLengthX10[iRoll, 1] :=
        UDINT_TO_WORD(udiHaiwellValueX10 MOD UDINT#65536);
END_FOR;


// Общий выпуск шестого энкодера: миллиметры -> 0,1 м.
uliHaiwellValueX10 := GVL_Runtime.stLineState.uliTotalLengthMm / ULINT#100;
IF uliHaiwellValueX10 <= ULINT#4294967295 THEN
    udiHaiwellValueX10 := ULINT_TO_UDINT(uliHaiwellValueX10);
ELSE
    udiHaiwellValueX10 := UDINT#4294967295;
    GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
    GVL_HAIWELL.wSystemStatusBits :=
        GVL_HAIWELL.wSystemStatusBits OR WORD#16#0020;
END_IF;
GVL_HAIWELL.a_wOutputTotalLengthX10[0] :=
    UDINT_TO_WORD(udiHaiwellValueX10 / UDINT#65536);
GVL_HAIWELL.a_wOutputTotalLengthX10[1] :=
    UDINT_TO_WORD(udiHaiwellValueX10 MOD UDINT#65536);

IF GVL_Runtime.stLineState.rSpeedMPerMin <= 0.0 THEN
    uiHaiwellSpeedX10 := UINT#0;
ELSIF GVL_Runtime.stLineState.rSpeedMPerMin >= 6553.5 THEN
    uiHaiwellSpeedX10 := UINT#65535;
ELSE
    uiHaiwellSpeedX10 :=
        REAL_TO_UINT(GVL_Runtime.stLineState.rSpeedMPerMin * 10.0);

        GVL_HAIWELL.wOutputStatus OR WORD#16#0001;

    GVL_HAIWELL.wOutputStatus :=
        GVL_HAIWELL.wOutputStatus OR WORD#16#0002;

IF GVL_Runtime.a_stHscState[6].xDataValid THEN
    GVL_HAIWELL.wOutputStatus :=
        GVL_HAIWELL.wOutputStatus OR WORD#16#0004;
END_IF;
IF GVL_Runtime.stLineState.xConfigFault THEN
    GVL_HAIWELL.wOutputStatus :=
        GVL_HAIWELL.wOutputStatus OR WORD#16#0010;

    GVL_SCADA.a_wQueueOverflowCount[0];
GVL_HAIWELL.a_wQueueOverflowCount[1] :=
    GVL_SCADA.a_wQueueOverflowCount[1];
GVL_HAIWELL.wCollectorHeartbeat := GVL_SCADA.wCollectorHeartbeat;
GVL_HAIWELL.wReserved2157 := WORD#0;

    (PersistentVars.uliShiftOutputPulses
     * UDINT_TO_ULINT(GVL_Constants.c_udiWheelCircumferenceMm))
    / UDINT_TO_ULINT(GVL_Constants.c_udiEncoderPulsesPerRevolution);

uliHaiwellValueX10 := uliShiftLengthMm / ULINT#100;

    udiHaiwellValueX10 := ULINT_TO_UDINT(uliHaiwellValueX10);
ELSE
    udiHaiwellValueX10 := UDINT#4294967295;
    GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
END_IF;
GVL_HAIWELL.a_wShiftOutputLengthX10[0] :=
    UDINT_TO_WORD(udiHaiwellValueX10 / UDINT#65536);
GVL_HAIWELL.a_wShiftOutputLengthX10[1] :=
    UDINT_TO_WORD(udiHaiwellValueX10 MOD UDINT#65536);


// Суммарный расход сырья за смену по разрешённым раскатам.
uliHaiwellValueX10 := uliShiftRawTotalMm / ULINT#100;
IF uliHaiwellValueX10 <= ULINT#4294967295 THEN
    udiHaiwellValueX10 := ULINT_TO_UDINT(uliHaiwellValueX10);
ELSE
    udiHaiwellValueX10 := UDINT#4294967295;
    GVL_Diagnostics.stSystem.xAnyConfigFault := TRUE;
END_IF;
GVL_HAIWELL.a_wShiftRawTotalLengthX10[0] :=
    UDINT_TO_WORD(udiHaiwellValueX10 / UDINT#65536);
GVL_HAIWELL.a_wShiftRawTotalLengthX10[1] :=
    UDINT_TO_WORD(udiHaiwellValueX10 MOD UDINT#65536);


// Средняя скорость выпуска рассчитывается за фактически прошедшее время
// текущего сменного интервала, а не как среднее мгновенных отсчётов.
rShiftAverageSpeedMPerMin := 0.0;
IF GVL_Runtime.stShiftState.xRtcValid
   AND (GVL_Runtime.stShiftState.udiTimestampUtc
        > PersistentVars.udiShiftStartTimestampUtc) THEN

        GVL_Runtime.stShiftState.udiTimestampUtc
        - PersistentVars.udiShiftStartTimestampUtc;

    rShiftAverageSpeedMPerMin :=
        ULINT_TO_REAL(uliShiftLengthMm)
        * 60.0
        / UDINT_TO_REAL(udiShiftElapsedSeconds)
        / 1000.0;

    GVL_HAIWELL.wShiftAverageSpeedX10 := WORD#65535;

    GVL_HAIWELL.wShiftAverageSpeedX10 :=
        UINT_TO_WORD(REAL_TO_UINT(rShiftAverageSpeedMPerMin * 10.0));
